cmake_minimum_required(VERSION 3.20)
project(on1x_benchmark LANGUAGES C CXX)
get_filename_component(MULIB_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
add_executable(on1x-benchmark native/main.cpp native/bench_host.c)
target_compile_features(on1x-benchmark PRIVATE cxx_std_20 c_std_11)
target_include_directories(on1x-benchmark SYSTEM PRIVATE "${MULIB_ROOT}/mudsl")
target_compile_definitions(on1x-benchmark PRIVATE BENCH_ROOT="${CMAKE_CURRENT_SOURCE_DIR}" MULIB_ROOT="${MULIB_ROOT}")
target_compile_options(on1x-benchmark PRIVATE -Wall -Wextra -Wpedantic)
enable_testing()
add_test(NAME benchmark-mull-selftest COMMAND on1x-benchmark selftest)
add_test(NAME benchmark-analysis COMMAND on1x-benchmark analysis-test)
add_test(NAME benchmark-invalid-stage-totals COMMAND on1x-benchmark invalid-stage-test)
set_tests_properties(benchmark-invalid-stage-totals PROPERTIES WILL_FAIL TRUE)
add_test(NAME benchmark-legacy-comparison-rejected COMMAND on1x-benchmark compare
  "${CMAKE_CURRENT_SOURCE_DIR}/results/linux-amd64-2026-11-21.json"
  "${CMAKE_CURRENT_SOURCE_DIR}/results/linux-amd64-2026-11-21.json")
set_tests_properties(benchmark-legacy-comparison-rejected PROPERTIES WILL_FAIL TRUE)
add_test(NAME benchmark-native-host COMMAND on1x-benchmark hosttest "${CMAKE_CURRENT_BINARY_DIR}/hosttest")
add_test(NAME benchmark-invalid-json COMMAND on1x-benchmark decode "{\"duplicate\":1,\"duplicate\":2}")
add_test(NAME benchmark-invalid-number COMMAND on1x-benchmark decode "01")
add_test(NAME benchmark-invalid-config COMMAND on1x-benchmark generate --repeats 0)
set_tests_properties(benchmark-invalid-json benchmark-invalid-number benchmark-invalid-config PROPERTIES WILL_FAIL TRUE)
add_executable(on1x-benchmark-host-tests ../tests/test_bench_host.c native/bench_host.c)
target_compile_features(on1x-benchmark-host-tests PRIVATE c_std_11)
set_target_properties(on1x-benchmark-host-tests PROPERTIES C_EXTENSIONS OFF)
target_compile_options(on1x-benchmark-host-tests PRIVATE -Wall -Wextra -Wpedantic -Werror)
add_test(NAME benchmark-host-process-group COMMAND on1x-benchmark-host-tests "${CMAKE_CURRENT_BINARY_DIR}")
