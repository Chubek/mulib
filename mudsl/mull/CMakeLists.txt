cmake_minimum_required(VERSION 3.24)
# Also a standalone entry point: cmake -S mudsl/mull -B build/mull.
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR AND NOT TARGET MuDSL)
    project(MuLL VERSION 1.2.0 LANGUAGES CXX)
    include(CTest)
    set(MULL_BUILD_CLI ON CACHE BOOL "Build the MuLL command line")
    option(MULL_ENABLE_MULIB "Build the optional MuLIB wrapper" OFF)
    add_subdirectory(.. mudsl)
    return()
endif()

option(MULL_BUILD_CLI "Build the MuLL command line" ON)
option(MULL_BUILD_D "Build the MuDSL.d compiler worker when a D compiler is available" ON)
if(NOT MULL_BUILD_CLI)
    return()
endif()
find_package(Python3 3.9 COMPONENTS Interpreter REQUIRED)
set(_mull_bin "${CMAKE_CURRENT_BINARY_DIR}/bin")
file(MAKE_DIRECTORY "${_mull_bin}")
configure_file(cli/mull.py "${_mull_bin}/mull" @ONLY)
file(CHMOD "${_mull_bin}/mull" PERMISSIONS OWNER_READ OWNER_WRITE OWNER_EXECUTE GROUP_READ GROUP_EXECUTE WORLD_READ WORLD_EXECUTE)
add_executable(mull-cpp cli/worker.cpp)
target_link_libraries(mull-cpp PRIVATE MuDSL)
set_target_properties(mull-cpp PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${_mull_bin}")

set(_mull_native OFF)
if(CMAKE_SYSTEM_NAME STREQUAL "Linux" AND CMAKE_SYSTEM_PROCESSOR MATCHES "^(x86_64|amd64|AMD64)$")
    set(_mull_native ON)
    add_library(mull-mu3ac-cpp STATIC mu3ac/Runtime.cpp mu3ac/Main.cpp)
    target_link_libraries(mull-mu3ac-cpp PRIVATE MuDSL)
    set_target_properties(mull-mu3ac-cpp PROPERTIES ARCHIVE_OUTPUT_DIRECTORY "${_mull_bin}")
    install(TARGETS mull-mu3ac-cpp ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR}/mull)
endif()

set(_mull_d "")
set(_mull_d_runtime "")
if(MULL_BUILD_D)
    find_program(MULL_D_COMPILER NAMES ldc2 dmd gdc)
    if(MULL_D_COMPILER)
        set(_mull_d "${_mull_bin}/mull-d")
        set(_mull_root "${CMAKE_CURRENT_SOURCE_DIR}/..")
        file(GLOB _mull_d_sources CONFIGURE_DEPENDS
            "${CMAKE_CURRENT_SOURCE_DIR}/mullier/*.d" "${CMAKE_CURRENT_SOURCE_DIR}/native/*/Native.d")
        list(APPEND _mull_d_sources "${_mull_root}/MuDSL.d" "${CMAKE_CURRENT_SOURCE_DIR}/stdlib/Stdlib.d")
        set(_mull_d_version_flags)
        set(_mull_d_link_flags)
        set(_mull_d_bridge_dep)
        if(MULL_ENABLE_MULIB)
            list(APPEND _mull_d_sources "${CMAKE_CURRENT_SOURCE_DIR}/bridge/Mulib/Native.d"
                "${CMAKE_CURRENT_SOURCE_DIR}/bridge/Mulib/Bridge.d")
            set(_mull_d_bridge_dep mull-mulib-bridge)
            if(MULL_D_COMPILER MATCHES "gdc")
                set(_mull_d_version_flags -fversion=MULL_ENABLE_MULIB)
                set(_mull_d_link_flags -lpthread -ldl -lm)
            elseif(MULL_D_COMPILER MATCHES "ldc")
                set(_mull_d_version_flags -d-version=MULL_ENABLE_MULIB)
                set(_mull_d_link_flags -L-lpthread -L-ldl -L-lm)
            else()
                set(_mull_d_version_flags -version=MULL_ENABLE_MULIB)
                set(_mull_d_link_flags -L-lpthread -L-ldl -L-lm)
            endif()
            list(APPEND _mull_d_link_flags $<TARGET_FILE:mull-mulib-bridge>)
        endif()
        set(_mull_d_output "-of=${_mull_d}")
        if(MULL_D_COMPILER MATCHES "gdc")
            set(_mull_d_output -o "${_mull_d}")
        endif()
        add_custom_command(OUTPUT "${_mull_d}"
            COMMAND ${MULL_D_COMPILER} ${_mull_d_output} ${_mull_d_version_flags} "-I${_mull_root}"
                "${CMAKE_CURRENT_SOURCE_DIR}/cli/worker.d"
                "${CMAKE_CURRENT_SOURCE_DIR}/mu3ac/Mu3AC.d" "${CMAKE_CURRENT_SOURCE_DIR}/mu3ac/DefaultRules.d" ${_mull_d_sources} ${_mull_d_link_flags}
            DEPENDS cli/worker.d mu3ac/Mu3AC.d mu3ac/DefaultRules.d ${_mull_d_sources} ${_mull_d_bridge_dep} VERBATIM)
        add_custom_target(mull-d-build ALL DEPENDS "${_mull_d}")
        install(PROGRAMS "${_mull_d}" DESTINATION ${CMAKE_INSTALL_LIBEXECDIR}/mull)
        if(_mull_native AND NOT MULL_D_COMPILER MATCHES "gdc")
            set(_mull_d_runtime "${_mull_bin}/libmull-mu3ac-d.a")
            set(_mull_d_objects -op)
            if(MULL_D_COMPILER MATCHES "ldc")
                set(_mull_d_objects -oq)
            endif()
            add_custom_command(OUTPUT "${_mull_d_runtime}"
                COMMAND ${MULL_D_COMPILER} -lib ${_mull_d_objects} ${_mull_d_version_flags} "-of=${_mull_d_runtime}" "-I${_mull_root}"
                    "${CMAKE_CURRENT_SOURCE_DIR}/mu3ac/Runtime.d" "${CMAKE_CURRENT_SOURCE_DIR}/mu3ac/Main.d" ${_mull_d_sources}
                DEPENDS mu3ac/Runtime.d mu3ac/Main.d ${_mull_d_sources} VERBATIM)
            add_custom_target(mull-mu3ac-d-build ALL DEPENDS "${_mull_d_runtime}")
            install(FILES "${_mull_d_runtime}" DESTINATION ${CMAKE_INSTALL_LIBDIR}/mull)
        elseif(_mull_native)
            message(STATUS "GDC worker enabled; D native runtime currently requires LDC or DMD")
        endif()
    else()
        message(STATUS "No D compiler found; mull --engine=cpp remains available")
    endif()
endif()

set(_cpp_runtime "")
set(_cpp_runtime_install "")
if(_mull_native)
    set(_cpp_runtime "${_mull_bin}/${CMAKE_STATIC_LIBRARY_PREFIX}mull-mu3ac-cpp${CMAKE_STATIC_LIBRARY_SUFFIX}")
    set(_cpp_runtime_install "${CMAKE_INSTALL_LIBDIR}/mull/${CMAKE_STATIC_LIBRARY_PREFIX}mull-mu3ac-cpp${CMAKE_STATIC_LIBRARY_SUFFIX}")
endif()
set(_d_install "")
set(_d_runtime_install "")
if(_mull_d)
    set(_d_install "${CMAKE_INSTALL_LIBEXECDIR}/mull/mull-d")
endif()
if(_mull_d_runtime)
    set(_d_runtime_install "${CMAKE_INSTALL_LIBDIR}/mull/libmull-mu3ac-d.a")
endif()
set(_bridge "")
set(_bridge_install "")
if(MULL_ENABLE_MULIB)
    set(_bridge "$<TARGET_FILE:mull-mulib-bridge>")
    set(_bridge_install "${CMAKE_INSTALL_LIBDIR}/${CMAKE_STATIC_LIBRARY_PREFIX}mull-mulib-bridge${CMAKE_STATIC_LIBRARY_SUFFIX}")
endif()
file(GENERATE OUTPUT "${_mull_bin}/mull-config.json" CONTENT
"{\"kind\":\"build\",\"cpp\":\"$<TARGET_FILE:mull-cpp>\",\"d\":\"${_mull_d}\",\"stdlib\":\"${CMAKE_CURRENT_SOURCE_DIR}/stdlib\",\"cpp_runtime\":\"${_cpp_runtime}\",\"d_runtime\":\"${_mull_d_runtime}\",\"bridge\":\"${_bridge}\",\"cxx\":\"${CMAKE_CXX_COMPILER}\",\"dc\":\"${MULL_D_COMPILER}\"}\n")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/mull-config-install.json"
"{\"kind\":\"install\",\"cpp\":\"${CMAKE_INSTALL_LIBEXECDIR}/mull/mull-cpp\",\"d\":\"${_d_install}\",\"stdlib\":\"${CMAKE_INSTALL_DATADIR}/mull/stdlib\",\"cpp_runtime\":\"${_cpp_runtime_install}\",\"d_runtime\":\"${_d_runtime_install}\",\"bridge\":\"${_bridge_install}\",\"cxx\":\"${CMAKE_CXX_COMPILER}\",\"dc\":\"${MULL_D_COMPILER}\"}\n")
install(PROGRAMS "${_mull_bin}/mull" DESTINATION ${CMAKE_INSTALL_BINDIR})
install(TARGETS mull-cpp RUNTIME DESTINATION ${CMAKE_INSTALL_LIBEXECDIR}/mull)
install(FILES "${CMAKE_CURRENT_BINARY_DIR}/mull-config-install.json" DESTINATION ${CMAKE_INSTALL_LIBEXECDIR}/mull RENAME mull-config.json)
install(FILES cli/mull.1 DESTINATION ${CMAKE_INSTALL_MANDIR}/man1)
install(FILES cli/mull.5 DESTINATION ${CMAKE_INSTALL_MANDIR}/man5)
install(FILES cli/README.md DESTINATION ${CMAKE_INSTALL_DATADIR}/doc/MuDSL/mull/cli)
install(FILES mu3ac/README.md DESTINATION ${CMAKE_INSTALL_DATADIR}/doc/MuDSL/mull/mu3ac)

    add_executable(mull_mu3ac_tests ../tests/test_mu3ac.cpp)
    target_link_libraries(mull_mu3ac_tests PRIVATE MuDSL)
    if(NOT MSVC)

    add_test(NAME mu3ac_embedded_rules COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/mu3ac/rewrite/embed.py --check)
    set(_mull_cli_test_args --mull "${_mull_bin}/mull" --cc "${CMAKE_CXX_COMPILER}" --native ${_mull_native})
    if(MULL_ENABLE_MULIB)
        list(APPEND _mull_cli_test_args --mulib)
    endif()
    add_test(NAME mull_cli COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/../scripts/check_mull_cli.py ${_mull_cli_test_args})
endif()
