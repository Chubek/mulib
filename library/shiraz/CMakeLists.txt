add_library(mulib_shiraz_core_objects OBJECT core.c)
add_library(mulib_shiraz_win32_objects OBJECT runtime.c kernel32.c registry.c)
foreach(target mulib_shiraz_core_objects mulib_shiraz_win32_objects)
  target_link_libraries(${target} PRIVATE mulib)
  target_compile_definitions(${target} PRIVATE _POSIX_C_SOURCE=200809L)
  set_target_properties(${target} PROPERTIES POSITION_INDEPENDENT_CODE ON)
endforeach()
add_library(mulib_shiraz STATIC $<TARGET_OBJECTS:mulib_shiraz_core_objects>)
add_library(mulib_shiraz_shared SHARED $<TARGET_OBJECTS:mulib_shiraz_core_objects>)
set_target_properties(mulib_shiraz mulib_shiraz_shared PROPERTIES OUTPUT_NAME shiraz)
add_library(mulib_shiraz_win32 STATIC $<TARGET_OBJECTS:mulib_shiraz_win32_objects>)
add_library(mulib_shiraz_win32_shared SHARED $<TARGET_OBJECTS:mulib_shiraz_win32_objects>)
set_target_properties(mulib_shiraz_win32 mulib_shiraz_win32_shared PROPERTIES OUTPUT_NAME shiraz-win32)
target_link_libraries(mulib_shiraz_win32 PUBLIC mulib_shiraz)
target_link_libraries(mulib_shiraz_win32_shared PUBLIC mulib_shiraz_shared)
add_executable(shiraz cli.c)
add_executable(shiraz-reg reg.c)
foreach(target shiraz shiraz-reg)
  target_compile_definitions(${target} PRIVATE _POSIX_C_SOURCE=200809L)
  target_link_libraries(${target} PRIVATE mulib mulib_shiraz_win32)
endforeach()
target_compile_definitions(shiraz PRIVATE
  SHIRAZ_INCLUDE_DIR="${PROJECT_SOURCE_DIR}/include"
  SHIRAZ_LIBRARY_DIR="${CMAKE_CURRENT_BINARY_DIR}")
